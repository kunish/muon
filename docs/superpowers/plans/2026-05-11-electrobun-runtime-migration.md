# Electrobun Runtime Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a parallel Electrobun desktop runtime for Muon without changing the current Electron default runtime.

**Architecture:** Keep the renderer-facing `window.muonDesktop` contract stable. Electron continues to expose it through `electron/preload.ts`; Electrobun exposes the same shape through a browser preload built with `Electroview` and a typed RPC schema handled by a Bun main process.

**Tech Stack:** Vue 3, Vite, TypeScript, Electron, Electrobun 1.18.1, Bun runtime APIs, Vitest source/contract tests.

---

Create `scripts/dev-electrobun.sh`:

```bash
#!/bin/bash
set -euo pipefail

pids=()
names=()

start() {
  echo "Starting $1..."
  names+=("$1")
  shift
  "$@" &
  pids+=("$!")
}

cleanup() {
  trap - EXIT INT TERM
  for pid in "${pids[@]}"; do
    if kill -0 "$pid" 2>/dev/null; then
      kill "$pid" 2>/dev/null || true
    fi
  done
  for pid in "${pids[@]}"; do
    wait "$pid" 2>/dev/null || true
  done
}

wait_for_url() {
  local url="$1"
  for _ in $(seq 1 60); do
    if curl -fsS "$url" >/dev/null 2>&1; then
      return 0
    fi
    sleep 0.5
  done
  echo "Timed out waiting for $url"
  return 1
}

trap cleanup EXIT
trap 'cleanup; exit 130' INT
trap 'cleanup; exit 143' TERM

start "Vite renderer" pnpm exec vite --host 0.0.0.0 --port 1420
wait_for_url http://localhost:1420
start "Electrobun desktop" env MUON_ELECTROBUN_RENDERER_URL=http://localhost:1420 pnpm exec electrobun dev --watch

while true; do
  for index in "${!pids[@]}"; do
    pid="${pids[$index]}"
    if ! kill -0 "$pid" 2>/dev/null; then
      set +e
      wait "$pid"
      status=$?
      set -e
      echo "${names[$index]} exited with status $status"
      exit "$status"
    fi
  done
  sleep 1
done
```

```ts
import type { RPCSchema } from 'electrobun'

export type Base64String = string

export interface ElectrobunSerializedBody {
  kind: 'bytes' | 'text'
  value: Base64String | string
}

export interface ElectrobunFetchRequest {
  url: string
  init?: {
    body?: ElectrobunSerializedBody
    headers?: Array<[string, string]>
    method?: string
    redirect?: RequestRedirect
  }
}

export interface ElectrobunFetchResponse {
  body: Base64String
  headers: Array<[string, string]>
  status: number
  statusText: string
  url: string
}

export interface ElectrobunDialogFilter {
  name: string
  extensions: string[]
}

export interface ElectrobunDialogOpenOptions {
  multiple?: boolean
  filters?: ElectrobunDialogFilter[]
}

export interface ElectrobunDialogSaveOptions {
  defaultPath?: string
  filters?: ElectrobunDialogFilter[]
}

export interface ElectrobunDialogAskOptions {
  title?: string
  detail?: string
  kind?: 'info' | 'warning' | 'error'
  okLabel?: string
  cancelLabel?: string
}

export type MuonElectrobunRPC = {
  bun: RPCSchema<{
    requests: {
      fetch: {
        params: ElectrobunFetchRequest
        response: ElectrobunFetchResponse
      }
      readFile: {
        params: { path: string }
        response: { bytes: Base64String }
      }
      writeFile: {
        params: { bytes: Base64String, path: string }
        response: void
      }
      openPath: {
        params: { path: string }
        response: string
      }
      revealItemInDir: {
        params: { path: string }
        response: void
      }
      openUrl: {
        params: { url: string }
        response: void
      }
      openDialog: {
        params: ElectrobunDialogOpenOptions | undefined
        response: string | string[] | null
      }
      saveDialog: {
        params: ElectrobunDialogSaveOptions | undefined
        response: string | null
      }
      askDialog: {
        params: { message: string, options?: ElectrobunDialogAskOptions }
        response: boolean
      }
      hideWindow: {
        params: void
        response: void
      }
      showWindow: {
        params: void
        response: void
      }
      focusWindow: {
        params: void
        response: void
      }
      setAutoLaunch: {
        params: { enabled: boolean }
        response: void
      }
      setCloseToTray: {
        params: { enabled: boolean }
        response: void
      }
      safeStorageIsAvailable: {
        params: void
        response: boolean
      }
      safeStorageEncrypt: {
        params: { plaintext: string }
        response: string
      }
      safeStorageDecrypt: {
        params: { base64: string }
        response: string
      }
      updaterCheck: {
        params: void
        response: { version: string } | null
      }
      updaterInstall: {
        params: void
        response: void
      }
    }
    messages: Record<never, never>
  }>
  webview: RPCSchema<{
    requests: Record<never, never>
    messages: {
      authCallback: { url: string }
    }
  }>
}
```

Create `src/electrobun/main.ts`:

```ts
import { Buffer } from 'node:buffer'
import { readFile, writeFile } from 'node:fs/promises'
import process from 'node:process'
import { BrowserView, BrowserWindow, Utils } from 'electrobun/bun'
import type {
  ElectrobunDialogAskOptions,
  ElectrobunDialogOpenOptions,
  ElectrobunFetchRequest,
  ElectrobunFetchResponse,
  ElectrobunSerializedBody,
  MuonElectrobunRPC,
} from './shared'

let mainWindow: BrowserWindow | null = null
let closeToTrayEnabled = false

function getMainWindow(): BrowserWindow {
  if (!mainWindow)
    throw new Error('Main window is not available')

  return mainWindow
}

function getRendererUrl(): string {
  return process.env.MUON_ELECTROBUN_RENDERER_URL || 'views://main/index.html'
}

function validateFilePath(filePath: string): void {
  if (!filePath || typeof filePath !== 'string')
    throw new Error('Invalid file path')

  if (filePath.includes('\0'))
    throw new Error('Invalid file path')
}

function isValidExternalUrl(value: string): boolean {
  try {
    const parsed = new URL(value)
    return parsed.protocol === 'http:' || parsed.protocol === 'https:' || parsed.protocol === 'mailto:'
  }
  catch {
    return false
  }
}

function normalizeMessageBoxKind(kind: ElectrobunDialogAskOptions['kind'] | undefined): 'info' | 'warning' | 'error' | 'question' {
  if (kind === 'error')
    return 'error'
  if (kind === 'warning')
    return 'warning'
  if (kind === 'info')
    return 'info'
  return 'question'
}

function normalizeAllowedFileTypes(options: ElectrobunDialogOpenOptions | undefined): string {
  const extensions = options?.filters?.flatMap(filter => filter.extensions) ?? []
  return extensions.length > 0 ? extensions.join(',') : '*'
}

function deserializeBody(body: ElectrobunSerializedBody | undefined): BodyInit | undefined {
  if (!body)
    return undefined

  if (body.kind === 'bytes')
    return Buffer.from(body.value, 'base64')

  return String(body.value)
}

function normalizeFetchInit(init: ElectrobunFetchRequest['init'] = {}): RequestInit {
  const body = deserializeBody(init.body)
  const normalized: RequestInit = {
    headers: init.headers,
    method: init.method,
    redirect: init.redirect,
  }

  if (body !== undefined)
    normalized.body = body

  return normalized
}

async function fetchFromBun(request: ElectrobunFetchRequest): Promise<ElectrobunFetchResponse> {
  const response = await fetch(request.url, normalizeFetchInit(request.init))
  const body = Buffer.from(await response.arrayBuffer()).toString('base64')
  const headers: Array<[string, string]> = []
  response.headers.forEach((value, key) => headers.push([key, value]))

  return {
    body,
    headers,
    status: response.status,
    statusText: response.statusText,
    url: response.url,
  }
}

function unsupported(message: string): never {
  throw new Error(message)
}

const rpc = BrowserView.defineRPC<MuonElectrobunRPC>({
  maxRequestTime: 10000,
  handlers: {
    requests: {
      async fetch(request) {
        return await fetchFromBun(request)
      },
      async readFile({ path }) {
        validateFilePath(path)
        const bytes = await readFile(path)
        return { bytes: Buffer.from(bytes).toString('base64') }
      },
      async writeFile({ bytes, path }) {
        validateFilePath(path)
        await writeFile(path, Buffer.from(bytes, 'base64'))
      },
      openPath({ path }) {
        validateFilePath(path)
        return Utils.openPath(path) ? '' : `Unable to open path: ${path}`
      },
      revealItemInDir({ path }) {
        validateFilePath(path)
        Utils.showItemInFolder(path)
      },
      openUrl({ url }) {
        if (!isValidExternalUrl(url))
          throw new Error(`Blocked opening URL with disallowed protocol: ${url.slice(0, 50)}`)

        Utils.openExternal(url)
      },
      async openDialog(options) {
        const filePaths = await Utils.openFileDialog({
          allowedFileTypes: normalizeAllowedFileTypes(options),
          allowsMultipleSelection: Boolean(options?.multiple),
          canChooseDirectory: false,
          canChooseFiles: true,
          startingFolder: '~/',
        })
        const normalized = filePaths.filter(Boolean)

        if (normalized.length === 0)
          return null

        return options?.multiple ? normalized : normalized[0]!
      },
      saveDialog() {
        return null
      },
      async askDialog({ message, options }) {
        const result = await Utils.showMessageBox({
          buttons: [options?.okLabel || 'OK', options?.cancelLabel || 'Cancel'],
          cancelId: 1,
          defaultId: 0,
          detail: options?.detail,
          message,
          title: options?.title,
          type: normalizeMessageBoxKind(options?.kind),
        })
        return result.response === 0
      },
      hideWindow() {
        getMainWindow().hide()
      },
      showWindow() {
        getMainWindow().show()
      },
      focusWindow() {
        getMainWindow().activate()
      },
      setAutoLaunch() {
        return undefined
      },
      setCloseToTray({ enabled }) {
        closeToTrayEnabled = Boolean(enabled)
      },
      safeStorageIsAvailable() {
        return false
      },
      safeStorageEncrypt() {
        return unsupported('safeStorage encryption is not available in Electrobun phase 1')
      },
      safeStorageDecrypt() {
        return unsupported('safeStorage encryption is not available in Electrobun phase 1')
      },
      updaterCheck() {
        return null
      },
```

```ts
  handlers: {
    requests: {},
    messages: {
      authCallback({ url }) {
        for (const callback of authCallbacks)
          callback(url)
      },
    },
  },
})

const electroview = new Electroview({ rpc })
const bridgeRpc = electroview.rpc!
const authCallbacks = new Set<(url: string) => void>()

function arrayBufferToBase64(buffer: ArrayBuffer): string {
  let binary = ''
  const bytes = new Uint8Array(buffer)
  for (const byte of bytes)
    binary += String.fromCharCode(byte)

  return btoa(binary)
}

function base64ToArrayBuffer(base64: string): ArrayBuffer {
  const binary = atob(base64)
  const bytes = new Uint8Array(binary.length)
  for (let index = 0; index < binary.length; index += 1)
    bytes[index] = binary.charCodeAt(index)

  return bytes.buffer
}

function serializeFetchRequest(request: SerializedFetchRequest): ElectrobunFetchRequest {
  const body = request.init?.body
  return {
    init: {
      body: body
        ? {
            kind: body.kind,
            value: body.kind === 'bytes'
              ? arrayBufferToBase64(body.value as ArrayBuffer)
              : String(body.value),
          }
        : undefined,
      headers: request.init?.headers,
      method: request.init?.method,
      redirect: request.init?.redirect,
    },
    url: request.url,
  }
}

function deserializeFetchResponse(response: ElectrobunFetchResponse): SerializedFetchResponse {
  return {
    body: base64ToArrayBuffer(response.body),
    headers: response.headers,
    status: response.status,
    statusText: response.statusText,
    url: response.url,
  }
}

function bytesToArrayBuffer(bytes: Uint8Array | ArrayBuffer): ArrayBuffer {
  if (bytes instanceof ArrayBuffer)
    return bytes

  const copy = new Uint8Array(bytes.byteLength)
  copy.set(bytes)
  return copy.buffer
}

      authCallbacks.add(callback)
      return () => authCallbacks.delete(callback)
```
